# 세이브 스키마 (saves/player.json) — 단일 진실 출처

> `saves/player.json` 이 게임 상태의 **유일한 진실**이다. 의미 있는 행동마다 갱신한다(`CLAUDE.md` 2절).
> 신규 게임이면 이 스키마대로 새로 만든다. `saves/player.example.json`에 동작 예시가 있다.

## 최상위 구조

```json
{
  "버전": 1,
  "메타": {
    "생성_실제시각": "2035-06-01T09:00:00Z",   // ISO. 실제 저장 시각(현실)
    "마지막_저장_실제시각": "2035-06-01T09:00:00Z",
    "플레이타임_분": 0
  },
  "rng": { "seed": 1234567890, "counter": 0 },   // 결정론 PRNG. CLAUDE.md 3절.

  "플레이어": {
    "이름": "",
    "출신지": { "국가": "", "도시": "" },
    "면허_등급": "신인",            // 신인|정규|상급|마스터
    "성별표현": ""                  // 자유. 묘사용. 비워도 됨.
  },

  "현재_위치": { "국가": "일본", "도시": "도쿄", "장소": "다이토 포켓몬 센터" },
  "게임내_날짜": { "년": 2035, "월": 6, "일": 1, "시간대": "아침" },  // 시간대: 새벽|아침|낮|저녁|밤

  "파티": [ /* 개체 객체 0~6개. 아래 "개체 스키마" 참조. 0번이 선두 */ ],
  "박스": [ /* 개체 객체 다수. 보관용 */ ],

  "인벤토리": {                      // 아이템명 → 수량
    "몬스터볼": 5, "상처약": 3
  },
  "소지금": { "통화": "JPY", "금액": 30000 },
  "명성": 0,                         // 전 세계 통합 명성 점수

  "도감_진행": {
    "본_종": [],                     // 목격한 종 id 목록
    "잡은_종": []                    // 포획한 종 id 목록
  },

  "대회_전적": [
    // { "id":"JP-TYO-L-007", "날짜":{"년":2035,"월":6,"일":21}, "결과":"우승", "획득명성":50, "상금":50000 }
  ],
  "보유_트로피_배지": [],            // 대회/업적 보상

  "스토리_플래그": [],               // 예: "튜토리얼_완료", "멘토_사카모토_만남"

  "현재_전투": null,                 // 전투 중이면 진행 상태 스냅샷(아래), 아니면 null
  "현재_퀘스트": []                  // 진행 중 목표 메모(자유 형식 문자열/객체)
}
```

## 개체 스키마 (파티·박스·NPC 보유 포켓몬 공통)

```json
{
  "개체_uid": "mon-0001",            // 세이브 내 고유. 교환/진화해도 유지.
  "종_id": 6,
  "form": "기본",                    // 강화는 일시적이므로 평소엔 "기본"
  "별명": "리자",                    // 없으면 종 이름 사용
  "레벨": 36,
  "경험치": 46656,                   // 누적 exp (training.md 경험치 그룹표 기준)
  "성격": "고집",                    // 25종 중 1. 능력치 보정은 training.md 표.
  "성별": "수",                      // 수|암|무성
  "IV": { "HP":31,"공격":31,"방어":20,"특공":15,"특방":22,"스피드":31 },
  "EV": { "HP":0,"공격":252,"방어":0,"특공":0,"특방":4,"스피드":252 },
  "특성": "맹화",
  "지닌도구": null,                  // 예: "리자드나이트", "구애머리띠"
  "기술": [                          // 최대 4개. PP 포함.
    { "기술":"플레어드라이브", "현재PP":15, "최대PP":15 }
  ],
  "현재_HP": 121,                    // 최대치는 종족값+IV/EV/레벨로 계산(training.md)
  "상태이상": null,                  // null|독|맹독|화상|마비|잠듦|얼음 (+잠듦 잔여턴 등 부가)
  "친밀도": 70,                      // 0~255. 일부 진화·기술에 영향
  "포획정보": { "장소":"시칠리아", "날짜":{"년":2035,"월":5,"일":20}, "볼":"몬스터볼" }
}
```

## 진행 중 전투 스냅샷 (`현재_전투`)

```json
{
  "전투_id": "btl-0007",
  "시작_counter": 412,               // 이 전투 시작 시점의 rng.counter (재현용)
  "유형": "야생",                    // 야생|트레이너|대회
  "상대": { /* NPC id 또는 야생 개체 */ },
  "내_활성_index": 0,
  "상대_활성_index": 0,
  "턴": 1,
  "랭크변화": { "나": {}, "상대": {} },  // 능력치 랭크(-6~+6)
  "날씨": null, "필드": null,
  "메가_사용": false, "거다이맥스_사용": false,  // 배틀당 각 1회 제한 추적
  "로그": []
}
```

## 무결성 규칙

- 파티는 **최대 6마리**. 모두 기절하면 전투 패배 처리.
- `종_id`는 `data/pokemon/index.json`에 존재해야 한다.
- 갱신은 **변경 필드만** 수정한다. 큰 변동 전 `saves/player.backup.json`으로 직전 상태 백업.
- `버전`이 다르면 마이그레이션 후 진행(스키마 변경 시 `CLAUDE.md` 8절).
